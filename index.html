# Dose-response curve fitting using different mathematical models.
# Libraries used: NumPy, Matplotlib and SciPy.
# For academic use only - NOT clinical dosing advice.


# Import required libraries
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import curve_fit


# 1. DATASET
# Dose represents the drug concentration.
# Response represents the corresponding therapeutic response.
# NaN means that the value is missing.

dose = np.array([0.5, 1, 1.5, 2, 2.5, 3, 4, 5, 6, 7,
                 8, 9, 10, 12, 15, np.nan])

resp = np.array([5.2, 11.8, 18.4, 26.3, 32.1, 38.7, 46.2,
                 52.8, 57.6, 61.4, 64.1, 65.9, 67.2,
                 68.5, 69.1, 70.0])


# 2. REMOVE INVALID VALUES
# Create two empty lists to store valid data.
X = []
Y = []

# Check each dose and response value.
# Ignore missing values and doses that are zero or negative.
for i in range(len(dose)):
    if not np.isnan(dose[i]) and not np.isnan(resp[i]) and dose[i] > 0:
        X.append(dose[i])
        Y.append(resp[i])

# Convert the lists into NumPy arrays.
X = np.array(X)
Y = np.array(Y)

# Display the number of valid data points.
print("Valid data points:", len(X))


# 3. DEFINE MATHEMATICAL MODELS

# Linear model: Represents a straight-line relationship.
def linear(x, m, c):
    return m*x + c

# Quadratic model: Represents a curved or parabolic relationship.
def quadratic(x, a, b, c):
    return a*x*x + b*x + c

# Exponential model: Shows response increasing towards a maximum value.
def exponential(x, ymax, k):
    return ymax * (1 - np.exp(-k*x))

# Hill model: Represents a saturation-type dose-response relationship.
def hill(x, emax, ec50, n):
    return emax * x**n / (ec50**n + x**n)


# 4. FIT ALL MATHEMATICAL MODELS

# Store each model with its name and initial parameter values.
# These initial values help curve_fit start the fitting process.
models = [
    ("Linear", linear, [1, 1]),
    ("Quadratic", quadratic, [-0.3, 9, 0]),
    ("Exponential", exponential, [70, 0.3]),
    ("Hill", hill, [70, 3, 1.5])
]

# Variables to store the best-performing model.
best_r2 = -999
best_name = ""
best_func = None
best_params = None

print("\nModel Results")
print("-------------------------------")

# Run a loop to fit and evaluate each model.
for name, func, initial in models:

    # Find the best parameters for the model using curve fitting.
    params, _ = curve_fit(func, X, Y, p0=initial, maxfev=10000)

    # Calculate the predicted response values.
    predicted = func(X, *params)

    # Calculate the difference between actual and predicted values.
    error = Y - predicted

    # Calculate Mean Squared Error (MSE).
    # Lower MSE means better prediction accuracy on this dataset.
    mse = np.mean(error**2)

    # Calculate R-squared (R2).
    # A value closer to 1 generally indicates a better fit.
    total = np.sum((Y - np.mean(Y))**2)
    r2 = 1 - np.sum(error**2) / total

    # Display the results of the current model.
    print("\nModel:", name)
    print("MSE:", round(mse, 4))
    print("R2:", round(r2, 4))
    print("Parameters:", np.round(params, 4))

    # Compare the current R2 with the best R2.
    # If it is higher, store this model as the best model.
    if r2 > best_r2:
        best_r2 = r2
        best_name = name
        best_func = func
        best_params = params

# Display the model with the highest R2.
print("\nBest Model:", best_name)


# 5. FIND DOSE FOR 60% RESPONSE USING EXPONENTIAL MODEL

# Set the target response to 60%.
target = 60

# Fit the exponential model to the dataset.
exp_params, _ = curve_fit(exponential, X, Y, p0=[70, 0.3])

# Extract the maximum response and growth rate.
ymax = exp_params[0]
k = exp_params[1]

# Calculate the required dose using the inverse exponential formula.
if target < ymax:
    required_dose = -np.log(1 - target/ymax) / k
    print("\nExponential model dose:", round(required_dose, 3), "mg/L")
else:
    print("\nTarget response cannot be reached by this model.")


# 6. CALCULATE DOSE USING THE BEST MODEL

# Initially, no dose has been found.
dose_found = None

# Check different dose values with a small interval of 0.001.
# Stop when the predicted response reaches or exceeds 60%.
for x in np.arange(X.min(), X.max(), 0.001):
    if best_func(x, *best_params) >= target:
        dose_found = x
        break

# Display the calculated dose.
if dose_found is not None:
    print("Best model dose:", round(dose_found, 3), "mg/L")
else:
    print("Target response not reached within the tested range.")


# 7. PLOT THE GRAPH

# Generate 300 equally spaced dose values for smooth curves.
xs = np.linspace(X.min(), X.max(), 300)

# Create a new graph with the specified size.
plt.figure(figsize=(9, 5))

# Plot the original measured data as black points.
plt.scatter(X, Y, color="black", label="Measured Data")

# Plot all four fitted mathematical models.
for name, func, initial in models:

    # Fit the current model again to get its parameters.
    params, _ = curve_fit(func, X, Y, p0=initial, maxfev=10000)

    # Highlight the exponential model using a red solid line.
    if name == "Exponential":
        plt.plot(xs, func(xs, *params), "r-", linewidth=3,
                 label="Exponential")
    else:
        # Display the remaining models using dashed lines.
        plt.plot(xs, func(xs, *params), "--",
                 label=name)

# Draw a horizontal line at the target response of 60%.
plt.axhline(target, color="gray", linestyle=":")

# Add graph title and axis labels.
plt.title("Dose vs Response Curve")
plt.xlabel("Dose Concentration (mg/L)")
plt.ylabel("Therapeutic Response (%)")

# Display the model names on the graph.
plt.legend()

# Display grid lines for easier reading.
plt.grid()

# Save the graph as an image file.
plt.savefig("calsympy_output.png")

print("\nGraph saved as calsympy_output.png")
print("WARNING: Academic analysis only - not clinical dosing advice.")

# Display the final graph.
plt.show()

1. The Exponential Model

We model the therapeutic response as a saturating exponential â€” a function that rises steeply at low doses, then flattens towards a ceiling:

\[ y = y_{\max}\!\left(1 - e^{-k x}\right) \]

where x is dose concentration (mg/L), y is therapeutic response (%), ymax is the maximum attainable response, and k is the rate constant controlling how quickly saturation is reached.

2. Data Cleaning

One dose entry is NaN. Invalid entries are removed before fitting using the condition:

Keep row i  if:  dose[i] â‰  NaN  and  resp[i] â‰  NaN  and  dose[i] > 0
â†’ 15 valid data points remain after filtering

3. Curve Fitting (Least-Squares)

Parameters \( y_{\max} \) and \( k \) are found by minimising the sum of squared residuals:

\[ \hat\theta = \arg\min_{\theta} \sum_{i=1}^{n} \!\left(y_i - f(x_i;\theta)\right)^2 \]

SciPy's curve_fit implements the Levenberg-Marquardt algorithm to solve this iteratively, starting from initial guesses \( y_{\max}^{(0)} = 70 \), \( k^{(0)} = 0.3 \). The fitted parameters are:

\[ y_{\max} = 74.052, \qquad k = 0.2352 \]

4. Mean Squared Error (MSE)

MSE measures the average squared distance between measured and predicted values. A lower MSE indicates a more accurate model.

\[ \text{MSE} = \frac{1}{n}\sum_{i=1}^{n}\!\left(y_i - \hat{y}_i\right)^2 \]

Substituting the exponential predictions:

\[ \text{MSE}_{\text{exp}} = \frac{1}{15}\sum_{i=1}^{15}\!\left(y_i - 74.052\!\left(1-e^{-0.2352\,x_i}\right)\right)^2 \approx \mathbf{0.14} \]

5. Coefficient of Determination (RÂ²)

RÂ² measures how much of the data's total variation the model explains. A value of 1 means a perfect fit.

\[ R^2 = 1 - \frac{SS_{\text{res}}}{SS_{\text{tot}}} \]

where

\[ SS_{\text{res}} = \sum_{i=1}^{n}\!\left(y_i - \hat{y}_i\right)^2, \qquad SS_{\text{tot}} = \sum_{i=1}^{n}\!\left(y_i - \bar{y}\right)^2 \]

For the exponential model:

\[ R^2_{\text{exp}} \approx 1 - \frac{2.10}{3680.4} \approx \mathbf{0.9994} \]

6. Finding the Dose for a 60 % Response

Rearranging the model equation to solve for dose x given a target response \( y^* \):

\[ y^* = y_{\max}\!\left(1 - e^{-kx}\right) \implies e^{-kx} = 1 - \frac{y^*}{y_{\max}} \] \[ \therefore\quad x = -\frac{1}{k}\ln\!\left(1 - \frac{y^*}{y_{\max}}\right) \]

Substituting \( y^* = 60 \), \( y_{\max} = 74.052 \), \( k = 0.2352 \):

\[ x = -\frac{1}{0.2352}\ln\!\left(1 - \frac{60}{74.052}\right) = -\frac{1}{0.2352}\ln(0.1803) \approx \mathbf{5.47\ \text{mg/L}} \]

7. Half-Life and Dosing Interval

For a first-order elimination model \( C(t) = C_0\,e^{-kt} \), the half-life is the time for concentration to halve:

\[ t_{1/2} = \frac{\ln 2}{k} = \frac{0.6931}{k} \]

With the fitted rate constant \( k = 0.2352 \):

\[ t_{1/2} = \frac{0.6931}{0.2352} \approx \mathbf{2.95\ \text{h}} \]

The dosing interval \( \tau \) required to keep concentration between a minimum \( C_{\min} \) and maximum \( C_{\max} \) is:

\[ \tau = \frac{1}{k}\ln\!\left(\frac{C_{\max}}{C_{\min}}\right) \]

8. Log-Linear Verification

Taking the natural logarithm of both sides of \( C(t) = C_0\,e^{-kt} \) gives:

\[ \ln C = \ln C_0 - k\,t \]

This is linear in \( t \) (slope \( = -k \), intercept \( = \ln C_0 \)). If \( \ln C \) plotted against \( t \) produces a straight line, the exponential model is confirmed.

9. Sensitivity â€” Partial Derivatives

Partial derivatives of \( C(t) = C_0\,e^{-kt} \) describe how sensitive the output is to each parameter:

\[ \frac{\partial C}{\partial C_0} = e^{-kt} \] \[ \frac{\partial C}{\partial k} = -C_0\,t\,e^{-kt} \]

At \( t = 4\text{ h} \), \( C_0 = 100 \), \( k = 0.2352 \):

\[ \frac{\partial C}{\partial C_0}\bigg|_{t=4} = e^{-0.9408} \approx 0.391 \] \[ \frac{\partial C}{\partial k}\bigg|_{t=4} = -100 \times 4 \times e^{-0.9408} \approx -156.3 \]

The large magnitude of \( \partial C/\partial k \) indicates the rate constant has the greatest influence on the predicted response.
